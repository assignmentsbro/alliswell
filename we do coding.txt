import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

df = pd.read_csv("Wine.csv")
print(df.head())

X = df.drop("Customer_Segment", axis=1)
y = df["Customer_Segment"]

X = StandardScaler().fit_transform(X)

pca = PCA(n_components=2)
result = pca.fit_transform(X)

print("PCA Result:\n", result)
print("Explained Variance:", pca.explained_variance_ratio_)
print("Total Variance:", sum(pca.explained_variance_ratio_))
print("Original Features:", X.shape[1])
print("Reduced Features:", result.shape[1])

plt.scatter(result[:, 0], result[:, 1], c=y)
plt.xlabel("PC1")
plt.ylabel("PC2")
plt.title("PCA")
plt.show()

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression, Ridge, Lasso
from sklearn.metrics import r2_score, mean_squared_error

df = pd.read_csv("uber.csv").dropna()
df = df.drop(columns=["key", "pickup_datetime", "Unnamed: 0"], errors="ignore")

q1, q3 = df.fare_amount.quantile([.25, .75])
df = df[df.fare_amount.between(q1-1.5*(q3-q1), q3+1.5*(q3-q1))]

df.corr()["fare_amount"].plot(kind="bar", title="Correlation")
plt.show()

X, y = df.drop("fare_amount", axis=1), df.fare_amount
a, b, c, d = train_test_split(X, y, test_size=.2, random_state=42)

for m in [LinearRegression(), Ridge(), Lasso()]:
    m.fit(a, c)
    p = m.predict(b)
    print(type(m).__name__, "R2:", r2_score(d, p),
          "RMSE:", mean_squared_error(d, p)**.5)
    plt.scatter(d, p)
    plt.xlabel("Actual")
    plt.ylabel("Predicted")
    plt.title(type(m).__name__)
    plt.show()

# SVM Handwritten Digit Classification

from sklearn.datasets import load_digits
from sklearn.model_selection import train_test_split
from sklearn.svm import SVC
import matplotlib.pyplot as plt

# Load data
digits = load_digits()
X_train, X_test, y_train, y_test = train_test_split(
    digits.data, digits.target, test_size=0.2, random_state=42)

# Train model
model = SVC()
model.fit(X_train, y_train)
pred = model.predict(X_test)

print("Accuracy:", model.score(X_test, y_test))

# Display digit images
fig, ax = plt.subplots(1, 5)
for i in range(5):
    ax[i].imshow(X_test[i].reshape(8, 8), cmap="gray")
    ax[i].set_title(f"Predicted: {pred[i]}")
    ax[i].axis("off")
plt.show()

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

df = pd.read_csv("Iris.csv")
X = df.iloc[:, 1:5]

# Elbow Method
wcss = [KMeans(n_clusters=k, n_init=10, random_state=42).fit(X).inertia_ for k in range(1, 11)]

plt.plot(range(1, 11), wcss, marker='o')
plt.title("Elbow Method")
plt.xlabel("Clusters")
plt.ylabel("WCSS")
plt.show()

# K-Means
df["Cluster"] = KMeans(n_clusters=3, n_init=10, random_state=42).fit_predict(X)

plt.scatter(X.iloc[:, 0], X.iloc[:, 1], c=df.Cluster, cmap="viridis")
plt.xlabel("Sepal Length")
plt.ylabel("Sepal Width")
plt.title("K-Means Clustering")
plt.show()

import pandas as pd
import matplotlib.pyplot as plt
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split
from sklearn.metrics import accuracy_score

df = pd.read_csv("car_evaluation.csv", header=None)
df.columns = ["buying","maint","doors","persons","lug_boot","safety","class"]

X = pd.get_dummies(df.drop("class", axis=1))
y = df["class"]

a,b,c,d = train_test_split(X,y,test_size=.2,random_state=42)

m = RandomForestClassifier().fit(a,c)
p = m.predict(b)

print("Accuracy:", accuracy_score(d,p))

y.value_counts().plot(kind="bar", title="Car Classes")
plt.show()

plt.scatter(d,p)
plt.xlabel("Actual")
plt.ylabel("Predicted")
plt.title("Car Classification")
plt.show()

import numpy as np
import matplotlib.pyplot as plt

maze = np.array([[0,0,0],[1,0,1],[0,0,2]])
Q = np.zeros((9,4))
actions = [-3,3,-1,1]

for _ in range(500):
    s = 0
    for _ in range(20):
        a = np.random.randint(4)
        n = s + actions[a]
        if n not in range(9) or maze.flat[n] == 1:
            continue
        r = 100 if n == 8 else -1
        Q[s,a] += .1*(r + .9*max(Q[n]) - Q[s,a])
        s = n
        if s == 8: break

plt.imshow(maze, cmap="viridis")
plt.title("Maze")
plt.show()
print("Training Completed!")
